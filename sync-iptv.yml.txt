name: Sync IPTV Source

on:
  schedule:
    - cron: '0 13 * * *'  # UTC 13:00 = 北京时间 21:00
  workflow_dispatch:

jobs:
  sync:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        with:
          persist-credentials: true

      - name: Update IPTV URLs
        run: |
          echo "=== 开始更新 IPTV 源 ==="
          echo "时间: $(date)"
          
          # 安装 ffmpeg
          echo "安装 ffmpeg..."
          sudo apt-get update -qq
          sudo apt-get install -y -qq ffmpeg
          
          # 转换换行符
          sed -i 's/\r$//' iptv.txt
          
          # 备份原文件
          cp iptv.txt iptv.txt.bak
          echo "原文件已备份"
          
          # 下载所有源并提取有效URL
          echo "下载并解析各源..."
          
          > /tmp/all_urls.txt
          
          curl -L --fail --connect-timeout 30 --max-time 60 \
            "https://raw.giteeusercontent.com/zhanghuan12345/my-live/raw/master/hbyd" 2>/dev/null | \
            grep -E '^[^#].+,' >> /tmp/all_urls.txt 2>/dev/null || true
          
          curl -L --fail --connect-timeout 30 --max-time 60 \
            "https://raw.githubusercontent.com/CCSH/IPTV/main/live.txt" 2>/dev/null | \
            grep -E '^[^#].+,' >> /tmp/all_urls.txt 2>/dev/null || true

          curl -L --fail --connect-timeout 30 --max-time 60 \
            "https://raw.githubusercontent.com/YueChan/Live/refs/heads/main/GNTV.m3u" 2>/dev/null | \
            grep -E '^[^#].+,' >> /tmp/all_urls.txt 2>/dev/null || true
          
          echo "下载完成，共获取 $(wc -l < /tmp/all_urls.txt) 条记录"
          
          # 标准化频道名函数
          normalize() {
            echo "$1" | sed 's/-//g' | sed 's/ //g' | tr 'A-Z' 'a-z'
          }
          
          # URL 有效性检测函数
          check_url() {
            url=$1
            timeout 3 ffprobe -v error -t 2 "$url" 2>/dev/null
            return $?
          }
          
          # 构建频道->URL映射（只取有效且最优的一个）
          > /tmp/url_map.txt
          
          grep -v '#genre#' iptv.txt | grep -v '^$' | sed 's/,$//' | while read template_channel; do
            [ -z "$template_channel" ] && continue
            
            template_norm=$(normalize "$template_channel")
            
            best_url=""
            best_priority=99
            
            while IFS=',' read -r source_channel source_url; do
              source_norm=$(normalize "$source_channel")
              
              # 模糊匹配
              if [ "$template_norm" = "$source_norm" ] || \
                 echo "$source_norm" | grep -q "$template_norm" || \
                 echo "$template_norm" | grep -q "$source_norm"; then
                
                # 优先级: http=1, rtsp=2, https=3
                priority=99
                if echo "$source_url" | grep -q '^http://'; then
                  priority=1
                elif echo "$source_url" | grep -q '^rtsp://'; then
                  priority=2
                elif echo "$source_url" | grep -q '^https://'; then
                  priority=3
                fi
                
                # 检测 URL 有效性
                if check_url "$source_url"; then
                  if [ $priority -lt $best_priority ]; then
                    best_priority=$priority
                    best_url="$source_url"
                  fi
                fi
              fi
            done < /tmp/all_urls.txt
            
            if [ -n "$best_url" ]; then
              # 只取第一个URL
              first_url=$(echo "$best_url" | cut -d'#' -f1)
              echo "$template_channel,$first_url" >> /tmp/url_map.txt
              echo "  ✅ $template_channel -> $first_url"
            else
              echo "  ⚠️ 未找到有效URL: $template_channel"
            fi
          done
          
          MATCH_COUNT=$(wc -l < /tmp/url_map.txt)
          echo "匹配完成，共匹配 $MATCH_COUNT 个频道"
          
          # 更新 iptv.txt
          echo "开始更新 iptv.txt..."
          > iptv.txt.new
          
          while IFS= read -r line; do
            if echo "$line" | grep -q '#genre#'; then
              echo "$line" >> iptv.txt.new
              continue
            fi
            
            if echo "$line" | grep -q '^[^#].\+,$'; then
              channel_name=$(echo "$line" | sed 's/,$//' | sed 's/^[[:space:]]*//;s/[[:space:]]*$//')
              
              url=$(grep "^$channel_name," /tmp/url_map.txt | head -1 | cut -d',' -f2-)
              
              if [ -n "$url" ]; then
                echo "$channel_name,$url" >> iptv.txt.new
              else
                echo "$channel_name," >> iptv.txt.new
              fi
              continue
            fi
            
            echo "$line" >> iptv.txt.new
          done < iptv.txt
          
          NEW_LINES=$(wc -l < iptv.txt.new)
          if [ "$NEW_LINES" -lt 50 ]; then
            echo "❌ 错误: 生成文件行数过少 ($NEW_LINES)"
            exit 1
          fi
          
          mv iptv.txt.new iptv.txt
          echo "✅ 更新完成"

      - name: Commit and Push
        run: |
          git config --local user.name "GitHub Action"
          git config --local user.email "action@github.com"
          git add iptv.txt
          
          if git diff --cached --quiet; then
            echo "✅ 内容无变化，跳过提交"
            exit 0
          fi
          
          git commit -m "Update IPTV URLs - $(date '+%Y-%m-%d %H:%M:%S')"
          git pull --rebase --strategy-option ours
          git push
          echo "✅ 已成功推送到仓库"
